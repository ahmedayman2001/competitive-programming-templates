bool check(int flr){
    cout<<"Current Floor is "<<flr<<"\n";

    int ans;    // 0->>egg break   ,  1->>egg not break ; //
    cin>>ans;

    return ans;
}

void go() {
    int n;
    cin >> n;  // Total number of floors
    int l = 1, r = n, mid, ans = -1;

    while (l <= r) {
        mid = (l + r) / 2;

        if (check(mid)) {  //   The egg will not break at this Floor ;  //

            ans = mid;
            l = mid + 1;


        } else {           //   The egg will break at this Floor ;  //


            r = mid - 1;

        }
    }

    cout << ans << "\n";
}
